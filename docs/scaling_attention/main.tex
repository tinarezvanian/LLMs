% tufte-book's \titleformat for \chapter only renders \thechapter as the
% chapter label (so an appendix would print just ``A''). Override here so
% the on-page chapter heading reads ``Appendix~A: <title>''. We keep
% \thechapter as just the letter so the TOC's \numberline slot, \ref{...},
% and page headers are unaffected.
\makeatletter
\titleformat{\chapter}%
  [display]%
  {\relax\ifthenelse{\NOT\boolean{@tufte@symmetric}}{\begin{fullwidth}}{}}%
  {\itshape\huge Appendix~\thechapter}%
  {0pt}%
  {\huge\rmfamily\itshape}%
  [\ifthenelse{\NOT\boolean{@tufte@symmetric}}{\end{fullwidth}}{}]
\makeatother
