% Portable preamble: pdfLaTeX + Latin Modern (no fontspec / tufte / marginnote).
% Inspired by Algorithmic_Adventures/preamble.tex (tufte + EB Garamond path).

\documentclass[11pt,twoside,openright]{book}

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{lmodern}
\usepackage[
    letterpaper,
    inner=1in,
    outer=1in,
    top=1in,
    bottom=1in,
    marginparwidth=1.5in,
    marginparsep=0.1in
]{geometry}

\usepackage{amsmath, amssymb, amsfonts}
\usepackage{graphicx}
\usepackage{microtype}
\usepackage{booktabs, xspace}
\usepackage[svgnames]{xcolor}
\usepackage{listings}
\usepackage{url}

\usepackage{hyperref}
\hypersetup{colorlinks=true,linkcolor=DarkSlateGray,urlcolor=DarkSlateGray,citecolor=DarkSlateGray}

\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5\baselineskip}
\sloppy
\setlength{\emergencystretch}{3em}

\setcounter{secnumdepth}{2}
\setcounter{tocdepth}{2}

\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries}{\chaptername\ \thechapter}{20pt}{\Huge}
\titleformat{\section}
  {\normalfont\Large\bfseries}{\thesection}{1em}{}
\titleformat{\subsection}
  {\normalfont\large\bfseries}{\thesubsection}{1em}{}

\lstset{
  basicstyle=\ttfamily\small,
  frame=single,
  breaklines=true,
  breakatwhitespace=false,
  postbreak=\mbox{\textcolor{gray}{$\hookrightarrow$}\space},
  showstringspaces=false,
  tabsize=2,
  language=Python,
}

\newcommand{\bigO}{\mathcal{O}}

\makeatletter
\def\maketitle{%
  \cleardoublepage
  \thispagestyle{empty}
  \null\vfill
  \begin{center}
    {\LARGE\bfseries\@title\par}
    \vspace{2em}
    {\large\itshape\@author\par}
    \vspace{2em}
    {\normalsize\@date\par}
  \end{center}
  \vfill\null
  \clearpage
}
\makeatother
